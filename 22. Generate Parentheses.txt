// 22. Generate Parentheses

class Solution {
    public List<String> generateParenthesis(int n) {
        List<String> ans = new ArrayList<>();
        backtrack(ans, new StringBuilder(), 0, 0, n);
        return ans;
    }

    public void backtrack(
        List<String> ans,
        StringBuilder str,
        int open,
        int closed,
        int n
    ) {
        if(str.length() == 2*n) {
            ans.add(str.toString());
            return;
        }

        if(open < n) {
            str.append("(");
            backtrack(ans, str, open + 1, closed, n);
            str.deleteCharAt(str.length() - 1);
        }

        if(open > closed) {
            str.append(")");
            backtrack(ans, str, open, closed + 1, n);
            str.deleteCharAt(str.length() - 1);
        }
    }
}
