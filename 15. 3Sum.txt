// 15. 3Sum

class Solution {
    public List<List<Integer>> threeSum(int[] nums) {
        Arrays.sort(nums);
        List<List<Integer>> res = new ArrayList<>();

        // once sorted, we must traverse the array
        for(int i = 0; i < nums.length; i++) {
            int a = nums[i]; // get first element of triplets
            // base cases
            if( a > 0 ) {
                // we break, as the first number is smallest
                // and therefore must be <= 0 for a triplet with 0 sum to be formed
                break;
            }

            if(i > 0 && a == nums[i - 1]) {
                // we skip this number as this would cause duplicates
                continue;
            }

            // main - once we've retrieved a usable 'a'
            // execute a 2sum using 2 pointer technique
            twoSum(nums, i, res);
        }

        return res;
    }

    public void twoSum(int[] nums, int i, List<List<Integer>> res) {
        int l = i + 1, r = nums.length - 1;

        while(l < r) {
            int sum = nums[i] + nums[l] + nums[r];
            if(sum < 0) {
                l++;
            } else if(sum > 0) {
                r--;
            } else {
                res.add(Arrays.asList(nums[i], nums[l++], nums[r--]));
                while(l < r && nums[l] == nums[l - 1]) l++;
            }
        }
    }
}
